# Junk/cache/reproducible directories that are ALWAYS excluded, even inside
# a repo carved back in via borg-extra-includes.txt - borg-backup.sh puts
# this file first in the concatenated patterns list, ahead of extra-includes
# and the Repos blanket-exclude, specifically so these win regardless (a
# dirty repo's *source* is what's worth backing up, not its venv/
# node_modules - verified by dry-run: without this ordering, carving in one
# dirty repo pulled its entire venv/site-packages along with it).
#
# Uses `re:` (regex) patterns, not `fm:` shell-glob patterns: borg-backup.sh
# runs `borg create` from ~ with `.` as the source, so top-level entries
# appear in the tree as bare relative names ("Repos", "Downloads", not
# "/home/.../Repos"). `fm:**/Repos` requires a path segment before "Repos"
# to match, so it silently fails to match (and therefore fails to exclude)
# exactly the top-level case that matters most here - verified by dry-run,
# where 61% of ~/Repos leaked through with the naive fm: pattern. `(^|/)`
# matches both "at the very start of the path" and "after a slash", so it
# correctly excludes the directory whether it's a top-level child of ~ or
# nested deeper (e.g. node_modules several levels inside a repo).
- re:(^|/)\.cache(/|$)
- re:(^|/)[Cc]ache
- re:(^|/)[Dd]ownloads(/|$)
- re:(^|/)\.var(/|$)
- re:(^|/)\.local/share/Trash(/|$)
- re:(^|/)node_modules(/|$)
- re:(^|/)venv(/|$)
- re:(^|/)\.venv(/|$)

# Electron/Chromium app junk (Discord, Claude, BeeperTexts, VSCodium, dbgate,
# ...). `Cache`/`Code Cache` are already caught by the [Cc]ache rule above;
# these are the rest of the regenerable Chromium profile dirs. Local Storage,
# IndexedDB etc. are deliberately NOT excluded (logins/settings live there).
- re:(^|/)(GPUCache|DawnWebGPUCache|DawnGraphiteCache|GraphiteDawnCache|GrShaderCache|ShaderCache|Crashpad|blob_storage|transcoded_cache)(/|$)
- re:^\.config/[^/]+/logs(/|$)
# Anything with "cache" ANYWHERE in its name, but only under top-level dotdirs
# (~/.config, ~/.local, ~/.mozilla, ...): catches "Code Cache", webcache,
# shaderhitcache, librarycache, discord_asset_cache, CachedProfilesData, etc.
# that the starts-with-cache rule above misses. Deliberately NOT applied to
# visible dirs (Documents, Pictures, ...) so a user folder with "cache" in its
# name isn't silently dropped.
- re:^\.[^/]+/(.*/)?[^/]*[Cc]ache[^/]*(/|$)
- re:(^|/)(Crash Reports|crashlogs|crashdumps)(/|$)
# GOG Galaxy backup folder: its own web cache/logs, all regenerated by Galaxy
- re:^gog-galaxy-backup-[^/]+/.*/Galaxy/(webcache|logs|crashdumps|autoupdate-verified)(/|$)
# Wine/Heroic prefixes: installer package cache and temp dirs
- re:^Games/.*/drive_c/ProgramData/Package Cache(/|$)
- re:^Games/.*/drive_c/users/[^/]+/(Temp|AppData/Local/Temp)(/|$)
- re:^Games/.*/drive_c/windows/temp(/|$)
- re:^Games/.*/AppData/Local/Microsoft/Windows/INetCache(/|$)
- re:^Games/Heroic/Prefixes/[^/]+/shadercache(/|$)
- re:^gog-galaxy-backup-[^/]+/.*/Galaxy/jump-list-cache(/|$)
- re:^\.local/share/Steam/steamapps/(workshop/)?temp(/|$)
- re:^\.thunderbird/[^/]+/crashes(/|$)
# Discord keeps one full copy of itself per version (app-1.0.159, ...)
- re:^\.config/discord/app-[0-9.]+(/|$)
# Claude desktop's downloaded Claude Code binaries
- re:^\.config/Claude/claude-code(/|$)

# Steam client files - the client re-downloads all of this on first launch.
# Kept on purpose: userdata/, steamapps/, config/config.vdf (and the other
# small files in config/), compatibilitytools.d/, local.vdf.
- re:^\.local/share/Steam/(appcache|bin|clientui|controller_base|depotcache|graphics|legacycompat|linux32|linux64|logs|package|public|resource|steam|steamrt32|steamrt64|steamui|tenfoot|ubuntu12_32|ubuntu12_64)(/|$)
- re:^\.local/share/Steam/config/htmlcache(/|$)

# Epub library: only ~/epubLibrary (merged-library + .git) is kept. The zip and
# ~/Documents/epubLibrary (the two original Calibre libraries) are fully
# contained in it or its git history; the 2 Mirrorshades epubs that weren't
# were copied into ~/epubLibrary/recovered-from-documents.
- re:^epubLibrary\.zip$
- re:^Documents/epubLibrary(/|$)

# Thunderbird IMAP mirrors (both accounts are Gmail/imap.gmail.com, so Google
# holds the real copy; Local Folders is ~16K and is still kept). Profile
# settings, address books and filters are kept.
- re:^\.thunderbird/[^/]+/ImapMail(/|$)

# Wine prefixes - rebuilt rather than restored (Heroic's and Lutris's shared
# prefix conflict is what corrupted BG3 before; start clean after the wipe).
# NOTE: this drops any game saves living inside a prefix.
# Exception: the Blackwell saves in the default prefix. borg doesn't recurse
# into an excluded dir, so the `+` rule has to cover the ancestor dirs too
# (dirs only - nothing else inside them matches), then everything else under
# Prefixes is excluded.
+ re:^Games/Heroic/Prefixes(/default(/drive_c(/users(/rockiesmagicnumber(/Saved Games(/.*)?)?)?)?)?)?$
- re:^Games/Heroic/Prefixes/.
- re:^\.wine(/|$)

# Runtimes/toolchains - reinstallable via rustup/nvm/npm/lutris/umu/claude.
# .cargo/bin and the nvm scripts themselves are small and are still kept, so
# the `. ~/.cargo/env` / nvm lines in .zshrc keep working after a restore.
- re:^\.rustup(/|$)
- re:^\.cargo/(registry|git)(/|$)
- re:^\.nvm/versions(/|$)
- re:^\.npm(/|$)
- re:^\.local/share/lutris/(runners|runtime)(/|$)
- re:^\.local/share/umu(/|$)
- re:^\.local/share/claude/versions(/|$)

# No trailing catch-all needed: patterns-from defaults to including anything
# that didn't match an exclude rule (verified by dry-run) - an earlier draft
# had a `+ pp:/home/rockiesmagicnumber` line here, but pp: matches against
# the archived (relative) path, so an absolute-looking prefix like that
# would never actually match anything. The blanket "- re:(^|/)Repos(/|$)"
# exclude (with the extra-includes carve-out) is appended by
# borg-backup.sh itself, after this file and after borg-extra-includes.txt -
# see that script for why the ordering matters.
